function defaultData() {
  return {
    servers: {},
    invites: {},
    cooldowns: {},
    blockedBumps: {},
    bumpReminders: {},
    languages: {},
    configs: {},

    ai: {
      channels: {},
      conversations: {},
      memberships: {},
      pendingInvites: {}
    }
  };
}

function mergeData(base, incoming) {
  for (const [key, value] of Object.entries(incoming || {})) {
    if (
      value &&
      typeof value === "object" &&
      !Array.isArray(value) &&
      base[key] &&
      typeof base[key] === "object" &&
      !Array.isArray(base[key])
    ) {
      base[key] =
        mergeData(base[key], value);
    } else {
      base[key] = value;
    }
  }

  return base;
}

function load() {
  try {
    if (!fs.existsSync(DATA_FILE)) {
      return defaultData();
    }

    const parsed =
      JSON.parse(
        fs.readFileSync(
          DATA_FILE,
          "utf8"
        )
      );

    return mergeData(
      defaultData(),
      parsed
    );
  } catch (error) {
    console.error(
      "❌ Impossible de lire data.json :",
      error
    );

    return defaultData();
  }
}

let saveQueue = Promise.resolve();

function save(data) {
  const snapshot =
    JSON.stringify(
      data,
      null,
      2
    );

  saveQueue =
    saveQueue
      .catch(() => {})
      .then(async () => {
        const dir =
          path.dirname(DATA_FILE);

        await fs.promises.mkdir(
          dir,
          {
            recursive: true
          }
        );

        const tempFile =
          `${DATA_FILE}.${process.pid}.${Date.now()}.${Math.random()
            .toString(36)
            .slice(2)}.tmp`;

        await fs.promises.writeFile(
          tempFile,
          snapshot,
          "utf8"
        );

        try {
          await fs.promises.rename(
            tempFile,
            DATA_FILE
          );
        } catch (error) {
          if (
            error.code === "EPERM" ||
            error.code === "EACCES"
          ) {
            await new Promise(
              resolve =>
                setTimeout(resolve, 100)
            );

            await fs.promises.rename(
              tempFile,
              DATA_FILE
            );
          } else {
            throw error;
          }
        }
      });

  return saveQueue;
}

function sleep(ms) {
  return new Promise(
    resolve => setTimeout(resolve, ms)
  );
}

function isAdmin(interaction) {
  return !!interaction.member
    ?.permissions
    ?.has(
      PermissionFlagsBits.Administrator
    );
}

function parseBoolean(value) {
  return /^(oui|yes|si|ja|true|1|on)$/i.test(
    String(value || "").trim()
  );
}

function langOf(data, guildId) {
  return LOCALES[
    data.languages[guildId]
  ]
    ? data.languages[guildId]
    : "fr";
}

function t(
  data,
  guildId,
  key,
  vars = {}
) {
  const lang =
    langOf(data, guildId);

  let text =
    LOCALES[lang]?.text?.[key] ||
    LOCALES.fr?.text?.[key] ||
    key;

  for (
    const [keyName, value]
    of Object.entries(vars)
  ) {
    text =
      text.replaceAll(
        `{${keyName}}`,
        String(value)
      );
  }

  return text;
}

async function safeReply(
  interaction,
  payload
) {
  try {
    if (
      interaction.replied ||
      interaction.deferred
    ) {
      return await interaction.followUp(payload);
    }

    return await interaction.reply(payload);
  } catch (error) {
    console.error(
      "❌ Réponse interaction impossible :",
      error?.message || error
    );
  }
}

function adminReply(
  interaction,
  data
) {
  return safeReply(
    interaction,
    {
      content:
        t(
          data,
          interaction.guildId,
          "adminOnly"
        ),
      ephemeral: true
    }
  );
}

function ensureAI(data) {
  if (!data.ai) {
    data.ai = {};
  }

  if (!data.ai.channels) {
    data.ai.channels = {};
  }

  if (!data.ai.conversations) {
    data.ai.conversations = {};
  }

  if (!data.ai.memberships) {
    data.ai.memberships = {};
  }

  if (!data.ai.pendingInvites) {
    data.ai.pendingInvites = {};
  }
}

function conversationIdForSolo(
  guildId,
  userId
) {
  return `solo:${guildId}:${userId}`;
}

function activeConversation(
  data,
  guildId,
  userId
) {
  ensureAI(data);

  return (
    data.ai.memberships[
      `${guildId}:${userId}`
    ] ||
    conversationIdForSolo(
      guildId,
      userId
    )
  );
}

function getConversation(
  data,
  id,
  guildId,
  ownerId
) {
  ensureAI(data);

  if (
    !data.ai.conversations[id]
  ) {
    data.ai.conversations[id] = {
      id,
      guildId,
      ownerId,
      members: [ownerId],
      messages: [],
      createdAt: Date.now(),
      updatedAt: Date.now()
    };
  }

  return data.ai.conversations[id];
}

function addAIMessage(
  conversation,
  role,
  userId,
  username,
  content
) {
  conversation.messages.push({
    role,
    userId,
    username,
    content,
    timestamp: Date.now()
  });

  if (
    conversation.messages.length >
    AI_MAX_MESSAGES
  ) {
    conversation.messages =
      conversation.messages.slice(
        -AI_MAX_MESSAGES
      );
  }

  conversation.updatedAt =
    Date.now();
}

function providerLabel(provider) {
  if (provider === "groq") {
    return "Groq";
  }

  return "Hugging Face";
}

async function callGroq(messages) {
  if (!groq) {
    throw new Error(
      "GROQ_API_KEY absent."
    );
  }

  const completion =
    await groq.chat.completions.create({
      model: GROQ_MODEL,
      messages,
      max_tokens: 1000,
      temperature: 0.7
    });

  const response =
    completion
      ?.choices?.[0]
      ?.message
      ?.content;

  if (
    !response ||
    typeof response !== "string"
  ) {
    throw new Error(
      "Réponse Groq vide."
    );
  }

  return response;
}

async function callHuggingFace(
  messages
) {
  if (!hf) {
    throw new Error(
      "HF_API_KEY absent."
    );
  }

  const completion =
    await hf.chatCompletion({
      model: HF_MODEL,
      messages,
      max_tokens: 1000,
      temperature: 0.7
    });

  const response =
    completion
      ?.choices?.[0]
      ?.message
      ?.content;

  if (
    !response ||
    typeof response !== "string"
  ) {
    throw new Error(
      "Réponse Hugging Face vide."
    );
  }

  return response;
}

async function generateAIResponse(
  conversation,
  user,
  messageText,
  imageUrls = [],
  guildName = null
) {
  const messages =
    buildMessages(
      conversation,
      imageUrls,
      guildName
    );

  const preferred =
    conversation.aiProvider ||
    DEFAULT_AI_PROVIDER;

  const providers =
    preferred === "huggingface"
      ? [
          [
            "huggingface",
            callHuggingFace
          ],
          [
            "groq",
            callGroq
          ]
        ]
      : [
          [
            "groq",
            callGroq
          ],
          [
            "huggingface",
            callHuggingFace
          ]
        ];

  let lastError;

  for (
    const [name, call]
    of providers
  ) {
    for (
      let attempt = 1;
      attempt <= 3;
      attempt++
    ) {
      try {
        const response =
          await call(messages);

        if (
          conversation.aiProvider &&
          conversation.aiProvider !== name
        ) {
          console.log(
            `🔄 IA utilisée temporairement : ${providerLabel(name)}`
          );
        }

        return response;
      } catch (error) {
        lastError = error;

        console.error(
          `❌ ${providerLabel(name)} - tentative ${attempt}/3 :`,
          error?.message || error
        );

        if (attempt < 3) {
          await sleep(
            attempt * 1000
          );
        }
      }
    }
  }

  console.error(
    "❌ ERREUR IA COMPLÈTE :",
    lastError
  );

  return (
    "> ```ERREUR TEMPORAIRE :```L'IA rencontre actuellement une difficulté temporaire. " +
    "Réessaie dans quelques secondes."
  );
}

const aiQueues =
  new Map();

function queueAI(
  conversationId,
  task
) {
  const previous =
    aiQueues.get(conversationId) ||
    Promise.resolve();

  const next =
    previous
      .catch(() => {})
      .then(task);

  aiQueues.set(
    conversationId,
    next
  );

  next.finally(() => {
    if (
      aiQueues.get(conversationId) === next
    ) {
      aiQueues.delete(
        conversationId
      );
    }
  });

  return next;
}

async function sendLong(
  channel,
  text,
  replyTo = null
) {
  const value =
    String(
      text ||
      "Je n'ai pas réussi à générer une réponse."
    );

  const chunks = [];

  for (
    let i = 0;
    i < value.length;
    i += 1900
  ) {
    chunks.push(
      value.slice(
        i,
        i + 1900
      )
    );
  }

  if (
    replyTo &&
    chunks.length
  ) {
    await replyTo.reply({
      content:
        chunks.shift()
    });
  }

  for (
    const chunk
    of chunks
  ) {
    await channel.send({
      content: chunk
    });
  }
}

function bumpRank(
  data,
  guildId
) {
  const sorted =
    Object.entries(
      data.servers
    )
      .sort(
        (a, b) =>
          Number(b[1]) -
          Number(a[1])
      );

  const index =
    sorted.findIndex(
      ([id]) =>
        id === guildId
    );

  return (
    index === -1
      ? null
      : index + 1
  );
}

const reminderTimers =
  new Map();

function clearBumpReminder(
  guildId
) {
  const timer =
    reminderTimers.get(guildId);

  if (timer) {
    clearTimeout(timer);
  }

  reminderTimers.delete(
    guildId
  );
}

function scheduleBumpReminder(
  guildId,
  expire
) {
  clearBumpReminder(guildId);

  const delay =
    Math.max(
      0,
      expire - Date.now()
    );

  const timer =
    setTimeout(
      async () => {
        try {
          const data = load();

          const cfg =
            data.bumpReminders[guildId];

          if (
            !cfg?.enabled ||
            !cfg?.channelId
          ) {
            return;
          }

          const channel =
            await client.channels
              .fetch(
                cfg.channelId
              )
              .catch(() => null);

          if (
            !channel?.isTextBased()
          ) {
            return;
          }

          const timestamp =
            Math.floor(
              expire / 1000
            );

          const text =
            (
              cfg.message ||
              "Le prochain bump est disponible <t:TIMESTAMP:R>."
            )
              .replaceAll(
                "TIMESTAMP",
                timestamp
              );

          await channel.send({
            content:
              `${cfg.mention || ""} ${text}`
                .trim()
          });
        } catch (error) {
          console.error(
            "❌ Erreur rappel-bump :",
            error
          );
        } finally {
          reminderTimers.delete(
            guildId
          );
        }
      },
      delay
    );

  reminderTimers.set(
    guildId,
    timer
  );
}

function restoreBumpReminders() {
  const data = load();

  for (
    const [guildId, last]
    of Object.entries(
      data.cooldowns
    )
  ) {
    const cfg =
      data.bumpReminders[guildId];

    const expire =
      Number(last) +
      COOLDOWN;

    if (
      cfg?.enabled &&
      expire > Date.now()
    ) {
      scheduleBumpReminder(
        guildId,
        expire
      );
    }
  }
}

  }
);

// =====================================================
// INTERACTIONS
// =====================================================

client.on(
  "interactionCreate",
  async interaction => {
    const data = load();

    const guildId =
      interaction.guildId;

    const userId =
      interaction.user?.id;

    try {
      // =================================================
      // COMMANDES
      // =================================================

      if (
        interaction.isChatInputCommand()
      ) {
        const cmd =
          interaction.commandName;

        const guildOnlyCommands = [
          "bump",
          "bump-invite",
          "top-bump",
          "top-bumps",
          "ia",
          "ai",
          "ki",
          "ia-invite",
          "ai-invite",
          "stats",
          "langue",
          "language",
          "idioma",
          "sprache",
          "config",
          "config-messages",
          "rappel-bump",
          "bump-reminder",
          "recordatorio-bump",
          "bump-erinnerung",
          "op-bump"
        ];

        if (
          !guildId &&
          guildOnlyCommands.includes(cmd)
        ) {
          return safeReply(
            interaction,
            {
              content:
                "❌ Cette commande doit être utilisée dans un serveur.",
              ephemeral: true
            }
          );
        }

        // -----------------------------------------------
        // /BUMP
        // -----------------------------------------------

        if (cmd === "bump") {
          if (
            data.blockedBumps[guildId]
          ) {
            return safeReply(
              interaction,
              {
                content:
                  "❌ Les bumps sont bloqués sur ce serveur.",
                ephemeral: true
              }
            );
          }

          if (
            !data.invites[guildId]
          ) {
            return safeReply(
              interaction,
              {
                content:
                  t(
                    data,
                    guildId,
                    "inviteMissing"
                  ),
                ephemeral: true
              }
            );
          }

          const now =
            Date.now();

          const last =
            Number(
              data.cooldowns[guildId] ||
              0
            );

          const expire =
            last + COOLDOWN;

          if (
            last &&
            now < expire
          ) {
            return safeReply(
              interaction,
              {
                embeds: [
                  new EmbedBuilder()
                    .setColor(0xff9900)
                    .setTitle(
                      "⏳ Bump indisponible"
                    )
                    .setThumbnail(
                      interaction.user
                        .displayAvatarURL()
                    )
                    .setDescription(
                      t(
                        data,
                        guildId,
                        "bumpWait",
                        {
                          timestamp:
                            Math.floor(
                              expire / 1000
                            )
                        }
                      )
                    )
                    .setImage(
                      IMAGE_URL
                    )
                ],
                ephemeral: true
              }
            );
          }

          data.cooldowns[guildId] =
            now;

          data.servers[guildId] =
            Number(
              data.servers[guildId] ||
              0
            ) + 1;

          await save(data);

          const nextExpire =
            now + COOLDOWN;

          scheduleBumpReminder(
            guildId,
            nextExpire
          );

          return safeReply(
            interaction,
            {
              embeds: [
                new EmbedBuilder()
                  .setColor(0x57f287)
                  .setTitle(
                    "✅ Bump effectué"
                  )
                  .setThumbnail(
                    interaction.user
                      .displayAvatarURL()
                  )
                  .setDescription(
                    `Merci à <@${userId}> d'avoir bumpé ce serveur.\n\n` +
                    `📊 Total : **${data.servers[guildId]} bumps**\n` +
                    `⏳ Prochain bump <t:${Math.floor(nextExpire / 1000)}:R>`
                  )
                  .setImage(
                    IMAGE_URL
                  )
              ]
            }
          );
        }

        // -----------------------------------------------
        // /BUMP-INVITE
        // -----------------------------------------------

        if (
          cmd === "bump-invite"
        ) {
          if (
            !isAdmin(interaction)
          ) {
            return adminReply(
              interaction,
              data
            );
          }

          const row =
            new ActionRowBuilder()
              .addComponents(
                new ChannelSelectMenuBuilder()
                  .setCustomId(
                    "select_bump_channel"
                  )
                  .setPlaceholder(
                    "Choisis le salon pour l'invitation"
                  )
                  .setChannelTypes(
                    ChannelType.GuildText
                  )
                  .setMinValues(1)
                  .setMaxValues(1)
              );

          return safeReply(
            interaction,
            {
              content:
                "🔗 Choisis le salon depuis lequel créer une invitation permanente.",
              components: [row],
              ephemeral: true
            }
          );
        }

        // -----------------------------------------------
        // /TOP-BUMP
        // -----------------------------------------------

        if (
          cmd === "top-bump" ||
          cmd === "top-bumps"
        ) {
          const sorted =
            Object.entries(
              data.servers
            )
              .sort(
                (a, b) =>
                  Number(b[1]) -
                  Number(a[1])
              )
              .slice(0, 30);

          let description = "";

          for (
            let index = 0;
            index < sorted.length;
            index++
          ) {
            const [
              serverId,
              bumps
            ] = sorted[index];

            const guild =
              client.guilds.cache.get(
                serverId
              );

            description +=
              `**#${index + 1}** ${
                guild
                  ? guild.name
                  : `Serveur ${serverId}`
              } — **${bumps} bumps**\n`;
          }

          const row =
            new ActionRowBuilder()
              .addComponents(
                new ButtonBuilder()
                  .setLabel(
                    "Voir plus"
                  )
                  .setEmoji(
                    "<:PlusBouton:1501339112420933803>"
                  )
                  .setStyle(
                    ButtonStyle.Link
                  )
                  .setURL(
                    "https://bp-discord.github.io/adminbot/bumps/top"
                  )
              );

          return safeReply(
            interaction,
            {
              embeds: [
                new EmbedBuilder()
                  .setColor(0xffcc00)
                  .setTitle(
                    "🏆 Top Bumps"
                  )
                  .setDescription(
                    description ||
                    t(
                      data,
                      guildId,
                      "topEmpty"
                    )
                  )
              ],
              components: [row]
            }
          );
        }

        // -----------------------------------------------
        // /IA
        // -----------------------------------------------

        if (
          cmd === "ia" ||
          cmd === "ai" ||
          cmd === "ki"
        ) {
          if (
            !isAdmin(interaction)
          ) {
            return adminReply(
              interaction,
              data
            );
          }

          const channel =
            interaction.options.getChannel(
              "salon",
              true
            );

          ensureAI(data);

          data.ai.channels[
            channel.id
          ] = true;

          ensureGuildConfig(
            data,
            guildId
          );

          await save(data);

          return safeReply(
            interaction,
            {
              content:
                t(
                  data,
                  guildId,
                  "iaEnabled",
                  {
                    channel:
                      `<#${channel.id}>`
                  }
                ),
              ephemeral: true
            }
          );
        }

        // -----------------------------------------------
        // /IA-INVITE
        // -----------------------------------------------

        if (
          cmd === "ia-invite" ||
          cmd === "ai-invite"
        ) {
          ensureAI(data);

          const target =
            interaction.options.getUser(
              "utilisateur",
              true
            );

          if (target.bot) {
            return safeReply(
              interaction,
              {
                content:
                  t(
                    data,
                    guildId,
                    "botInvite"
                  ),
                ephemeral: true
              }
            );
          }

          if (
            target.id === userId
          ) {
            return safeReply(
              interaction,
              {
                content:
                  t(
                    data,
                    guildId,
                    "selfInvite"
                  ),
                ephemeral: true
              }
            );
          }

          if (
            !interaction.guild ||
            data.ai.channels[
              interaction.channelId
            ] !== true
          ) {
            return safeReply(
              interaction,
              {
                content:
                  t(
                    data,
                    guildId,
                    "aiNotEnabled"
                  ),
                ephemeral: true
              }
            );
          }

          const convId =
            activeConversation(
              data,
              guildId,
              userId
            );

          const conv =
            getConversation(
              data,
              convId,
              guildId,
              userId
            );

          if (
            conv.members.includes(
              target.id
            )
          ) {
            return safeReply(
              interaction,
              {
                content:
                  t(
                    data,
                    guildId,
                    "alreadyMember"
                  ),
                ephemeral: true
              }
            );
          }

          const inviteKey =
            `${guildId}:${convId}:${target.id}`;

          if (
            data.ai.pendingInvites[
              inviteKey
            ]
          ) {
            return safeReply(
              interaction,
              {
                content:
                  "⏳ Une invitation est déjà en attente.",
                ephemeral: true
              }
            );
          }

          try {
            const dm =
              await target.createDM();

            const embed =
              new EmbedBuilder()
                .setColor(
                  0x5865f2
                )
                .setTitle(
                  "🤖 Invitation à une discussion IA"
                )
                .setDescription(
                  `**${interaction.user.username}** t'a invité à sa discussion sur **${interaction.guild.name}**.`
                );

            const buttons =
              new ActionRowBuilder()
                .addComponents(
                  new ButtonBuilder()
                    .setCustomId(
                      `aiinvite_accept|${guildId}|${convId}|${target.id}`
                    )
                    .setLabel(
                      "Accepter"
                    )
                    .setStyle(
                      ButtonStyle.Success
                    ),

                  new ButtonBuilder()
                    .setCustomId(
                      `aiinvite_decline|${guildId}|${convId}|${target.id}`
                    )
                    .setLabel(
                      "Refuser"
                    )
                    .setStyle(
                      ButtonStyle.Danger
                    )
                );

            await dm.send({
              embeds: [embed],
              components: [buttons]
            });

            data.ai.pendingInvites[
              inviteKey
            ] = {
              guildId,
              convId,
              inviterId: userId,
              targetId: target.id,
              createdAt: Date.now()
            };

            await save(data);

            return safeReply(
              interaction,
              {
                content:
                  t(
                    data,
                    guildId,
                    "aiInviteSent",
                    {
                      user:
                        `<@${target.id}>`
                    }
                  ),
                ephemeral: true
              }
            );
          } catch (error) {
            console.error(
              "❌ Impossible d'envoyer l'invitation IA :",
              error
            );

            return safeReply(
              interaction,
              {
                content:
                  t(
                    data,
                    guildId,
                    "mpFailed"
                  ),
                ephemeral: true
              }
            );
          }
        }

        // -----------------------------------------------
        // /STATS
        // -----------------------------------------------

        if (
          cmd === "stats"
        ) {
          const owner =
            await interaction.guild
              .fetchOwner()
              .catch(() => null);

          const rank =
            bumpRank(
              data,
              guildId
            );

          const embed =
            new EmbedBuilder()
              .setColor(0x5865f2)
              .setTitle(
                t(
                  data,
                  guildId,
                  "statsTitle",
                  {
                    guild:
                      interaction.guild.name
                  }
                )
              )
              .setThumbnail(
                interaction.guild.iconURL({
                  size: 1024
                })
              )
              .addFields(
                {
                  name:
                    t(
                      data,
                      guildId,
                      "statsBumps"
                    ),

                  value:
                    `**${
                      Number(
                        data.servers[guildId] ||
                        0
                      )
                    }**`,

                  inline: true
                },

                {
                  name:
                    t(
                      data,
                      guildId,
                      "statsOwner"
                    ),

                  value:
                    owner
                      ? `<@${owner.id}>`
                      : "—",

                  inline: true
                },

                {
                  name:
                    t(
                      data,
                      guildId,
                      "statsMembers"
                    ),

                  value:
                    String(
                      interaction.guild.memberCount
                    ),

                  inline: true
                },

                {
                  name:
                    t(
                      data,
                      guildId,
                      "statsRoles"
                    ),

                  value:
                    String(
                      interaction.guild.roles.cache.size
                    ),

                  inline: true
                },

                {
                  name:
                    t(
                      data,
                      guildId,
                      "statsChannels"
                    ),

                  value:
                    String(
                      interaction.guild.channels.cache.size
                    ),

                  inline: true
                },

                {
                  name:
                    t(
                      data,
                      guildId,
                      "statsRank"
                    ),

                  value:
                    rank
                      ? `#${rank}`
                      : "—",

                  inline: true
                }
              );

          return safeReply(
            interaction,
            {
              embeds: [embed]
            }
          );
        }

        // -----------------------------------------------
        // /LANGUE
        // -----------------------------------------------

        if (
          [
            "langue",
            "language",
            "idioma",
            "sprache"
          ].includes(cmd)
        ) {
          if (
            !isAdmin(interaction)
          ) {
            return adminReply(
              interaction,
              data
            );
          }

          const selected =
            interaction.options.getString(
              "langue",
              true
            );

          if (
            !LOCALES[selected]
          ) {
            return safeReply(
              interaction,
              {
                content:
                  "❌ Langue invalide.",
                ephemeral: true
              }
            );
          }

          data.languages[guildId] =
            selected;

          ensureGuildConfig(
            data,
            guildId
          );

          await save(data);

          return safeReply(
            interaction,
            {
              embeds: [
                new EmbedBuilder()
                  .setColor(0x5865f2)
                  .setTitle(
                    t(
                      data,
                      guildId,
                      "language"
                    )
                  )
                  .setDescription(
                    t(
                      data,
                      guildId,
                      "languageSaved",
                      {
                        language:
                          LOCALES[selected].name
                      }
                    )
                  )
              ],
              ephemeral: true
            }
          );
        }

        // -----------------------------------------------
        // /CONFIG
        // -----------------------------------------------

        if (
          cmd === "config" ||
          cmd === "config-messages"
        ) {
          if (
            !isAdmin(interaction)
          ) {
            return adminReply(
              interaction,
              data
            );
          }

          ensureGuildConfig(
            data,
            guildId
          );

          await save(data);

          return safeReply(
            interaction,
            {
              ...configResponse(
                data,
                interaction.guild
              ),
              ephemeral: true
            }
          );
        }

        // -----------------------------------------------
        // /RAPPEL-BUMP
        // -----------------------------------------------

        if (
          [
            "rappel-bump",
            "bump-reminder",
            "recordatorio-bump",
            "bump-erinnerung"
          ].includes(cmd)
        ) {
          if (
            !isAdmin(interaction)
          ) {
            return adminReply(
              interaction,
              data
            );
          }

          if (
            !data.bumpReminders[guildId]
          ) {
            data.bumpReminders[guildId] = {
              enabled: false,
              channelId: null,
              message:
                "Le prochain bump est disponible <t:TIMESTAMP:R>.",
              mention: ""
            };
          }

          const cfg =
            data.bumpReminders[guildId];

          const row =
            new ActionRowBuilder()
              .addComponents(
                new ButtonBuilder()
                  .setCustomId(
                    "reminder_toggle"
                  )
                  .setLabel(
                    cfg.enabled
                      ? "Désactiver"
                      : "Activer"
                  )
                  .setStyle(
                    cfg.enabled
                      ? ButtonStyle.Danger
                      : ButtonStyle.Success
                  ),

                new ButtonBuilder()
                  .setCustomId(
                    "reminder_channel"
                  )
                  .setLabel(
                    "Choisir le salon"
                  )
                  .setStyle(
                    ButtonStyle.Primary
                  ),

                new ButtonBuilder()
                  .setCustomId(
                    "reminder_message"
                  )
                  .setLabel(
                    "Modifier le message"
                  )
                  .setStyle(
                    ButtonStyle.Secondary
                  )
              );

          return safeReply(
            interaction,
            {
              embeds: [
                new EmbedBuilder()
                  .setTitle(
                    "⏰ Rappel-bump"
                  )
                  .setDescription(
                    `Système : **${
                      cfg.enabled
                        ? "Activé"
                        : "Désactivé"
                    }**\n` +
                    `Salon : ${
                      cfg.channelId
                        ? `<#${cfg.channelId}>`
                        : "Non configuré"
                    }\n` +
                    `Message : ${cfg.message}\n` +
                    `Mention : ${
                      cfg.mention ||
                      "Aucune"
                    }`
                  )
              ],
              components: [row],
              ephemeral: true
            }
          );
        }

        // -----------------------------------------------
        // /OP-BUMP
        // -----------------------------------------------

        if (
          cmd === "op-bump"
        ) {
          if (
            userId !==
            process.env.OWNER_ID
          ) {
            return safeReply(
              interaction,
              {
                content:
                  "❌ Cette commande est réservée au créateur d'AdminBot™.",
                ephemeral: true
              }
            );
          }

          const row =
            new ActionRowBuilder()
              .addComponents(
                new ButtonBuilder()
                  .setCustomId(
                    "op_set_bumps"
                  )
                  .setLabel(
                    "Set bumps"
                  )
                  .setStyle(
                    ButtonStyle.Primary
                  ),

                new ButtonBuilder()
                  .setCustomId(
                    "op_block_bump"
                  )
                  .setLabel(
                    "Blocked bump"
                  )
                  .setStyle(
                    ButtonStyle.Danger
                  ),

                new ButtonBuilder()
                  .setCustomId(
                    "op_config_bump"
                  )
                  .setLabel(
                    "Configure bump"
                  )
                  .setStyle(
                    ButtonStyle.Secondary
                  )
              );

          return safeReply(
            interaction,
            {
              embeds: [
                new EmbedBuilder()
                  .setColor(0x5865f2)
                  .setTitle(
                    "🛠️ OP-Bump"
                  )
                  .setDescription(
                    "Choisis une opération."
                  )
              ],
              components: [row],
              ephemeral: true
            }
          );
        }
      }

      // =================================================
      // CHANNEL SELECT
      // =================================================

      if (
        interaction.isChannelSelectMenu()
      ) {
        const id =
          interaction.customId;

        if (
          !isAdmin(interaction)
        ) {
          return adminReply(
            interaction,
            data
          );
        }

        const channel =
          interaction.channels.first();

        if (
          id === "select_bump_channel"
        ) {
          if (!channel) {
            return interaction.update({
              content:
                "❌ Salon invalide.",
              components: []
            });
          }

          const invite =
            await channel.createInvite({
              maxAge: 0,
              maxUses: 0,
              unique: true
            });

          data.invites[guildId] =
            invite.url;

          await save(data);

          return interaction.update({
            content:
              `🔗 Invite créée depuis <#${channel.id}>.`,
            components: []
          });
        }

        if (
          id === "config_welcome_channel"
        ) {
          ensureGuildConfig(
            data,
            guildId
          );

          data.configs[guildId]
            .welcome
            .channelId =
              channel?.id || null;

          await save(data);

          return interaction.update({
            content:
              `👋 Salon d'arrivée enregistré : ${
                channel
                  ? `<#${channel.id}>`
                  : "—"
              }`,
            components: []
          });
        }

        if (
          id === "config_leave_channel"
        ) {
          ensureGuildConfig(
            data,
            guildId
          );

          data.configs[guildId]
            .leave
            .channelId =
              channel?.id || null;

          await save(data);

          return interaction.update({
            content:
              `🚪 Salon de départ enregistré : ${
                channel
                  ? `<#${channel.id}>`
                  : "—"
              }`,
            components: []
          });
        }

        if (
          id === "reminder_channel"
        ) {
          if (
            !data.bumpReminders[guildId]
          ) {
            data.bumpReminders[guildId] = {
              enabled: false,
              channelId: null,
              message:
                "Le prochain bump est disponible <t:TIMESTAMP:R>.",
              mention: ""
            };
          }

          data.bumpReminders[guildId]
            .channelId =
              channel?.id || null;

          await save(data);

          return interaction.update({
            content:
              `⏰ Salon enregistré : ${
                channel
                  ? `<#${channel.id}>`
                  : "—"
              }`,
            components: []
          });
        }
      }

      // =================================================
      // ROLE SELECT
      // =================================================

      if (
        interaction.isRoleSelectMenu()
      ) {
        if (
          interaction.customId ===
          "config_roles_select"
        ) {
          if (
            !isAdmin(interaction)
          ) {
            return adminReply(
              interaction,
              data
            );
          }

          const cfg =
            ensureGuildConfig(
              data,
              guildId
            );

          cfg.roles.roleIds =
            [...interaction.values];

          cfg.roles.enabled =
            cfg.roles.roleIds.length > 0;

          await save(data);

          return interaction.update({
            content:
              `🎭 ${
                cfg.roles.roleIds.length
                  ? "Rôles enregistrés."
                  : "Aucun rôle sélectionné."
              }`,
            components: []
          });
        }
      }

      // =================================================
      // BOUTONS
      // =================================================

      if (
        interaction.isButton()
      ) {
        const id =
          interaction.customId;

        // Invitations IA.
        if (
          id.startsWith("aiinvite_")
        ) {
          const [
            action,
            inviteGuildId,
            convId,
            targetId
          ] =
            id.split("|");

          const inviteKey =
            `${inviteGuildId}:${convId}:${targetId}`;

          ensureAI(data);

          const invite =
            data.ai.pendingInvites[
              inviteKey
            ];

          const guildName =
            client.guilds.cache.get(
              inviteGuildId
            )?.name ||
            "ce serveur";

          if (
            action === "aiinvite_accept"
          ) {
            if (
              interaction.user.id !== targetId ||
              !invite
            ) {
              return interaction.update({
                content:
                  "❌ Cette invitation n'est plus valide.",
                embeds: [],
                components: []
              });
            }

            const conversation =
              getConversation(
                data,
                convId,
                inviteGuildId,
                invite.inviterId
              );

            if (
              !conversation.members.includes(
                targetId
              )
            ) {
              conversation.members.push(
                targetId
              );
            }

            data.ai.memberships[
              `${inviteGuildId}:${targetId}`
            ] = convId;

            delete data.ai.pendingInvites[
              inviteKey
            ];

            await save(data);

            const row =
              new ActionRowBuilder()
                .addComponents(
                  new ButtonBuilder()
                    .setCustomId(
                      `aiinvite_solo|${inviteGuildId}|${convId}|${targetId}`
                    )
                    .setLabel(
                      "Revenir en solo"
                    )
                    .setStyle(
                      ButtonStyle.Secondary
                    )
                );

            return interaction.update({
              embeds: [
                new EmbedBuilder()
                  .setColor(0x57f287)
                  .setDescription(
                    `Tu as accepté l'invitation à rejoindre la discussion sur **${guildName}**.`
                  )
              ],
              components: [row]
            });
          }

          if (

          return safeReply(
            interaction,
            {
              content:
                `✅ Invitation du serveur ${server} mise à jour.`,
              ephemeral: true
            }
          );
        }
      }
    } catch (error) {
      console.error(
        "❌ Interaction error :",
        error
      );

      if (
        !interaction.replied &&
        !interaction.deferred
      ) {
        await safeReply(
          interaction,
          {
            content:
              "❌ Une erreur interne est survenue.",
            ephemeral: true
          }
        );
      }
    }
  }
);

// ARRIVÉE MEMBRE

client.on(
  "guildMemberAdd",
  async member => {
    try {
      const data = load();

      const cfg =
        data.configs?.[
          member.guild.id
        ];

      if (!cfg) {
        return;
      }

      if (
        cfg.roles?.enabled &&
        Array.isArray(
          cfg.roles.roleIds
        )
      ) {
        for (
          const roleId
          of cfg.roles.roleIds
        ) {
          const role =
            member.guild.roles.cache.get(
              roleId
            );

          if (role) {
            await member.roles
              .add(role)
              .catch(() => {});
          }
        }
      }

      if (
        cfg.nicknames?.members?.enabled &&
        !member.user.bot
      ) {
        const nickname =
          cfg.nicknames.members.format
            .replaceAll(
              "{username}",
              member.user.username
            )
            .replaceAll(
              "{displayname}",
              member.displayName
            );

        await member.setNickname(
          nickname
        ).catch(() => {});
      }

      if (
        cfg.nicknames?.bots?.enabled &&
        member.user.bot
      ) {
        const nickname =
          cfg.nicknames.bots.format
            .replaceAll(
              "{username}",
              member.user.username
            )
            .replaceAll(
              "{displayname}",
              member.displayName
            );

        await member.setNickname(
          nickname
        ).catch(() => {});
      }

      if (
        cfg.welcome?.enabled &&
        cfg.welcome?.channelId
      ) {
        const channel =
          member.guild.channels.cache.get(
            cfg.welcome.channelId
          );

        if (
          channel?.isTextBased()
        ) {
          const text =
            cfg.welcome.message
              .replaceAll(
                "<@user>",
                `<@${member.id}>`
              )
              .replaceAll(
                "{user}",
                `<@${member.id}>`
              )
              .replaceAll(
                "{username}",
                member.user.username
              )
              .replaceAll(
                "{server}",
                member.guild.name
              );

          await channel.send({
            content: text
          });
        }
      }
    } catch (error) {
      console.error(
        "❌ Erreur arrivée membre :",
        error
      );
    }
  }
);

client.on(
  "guildMemberRemove",
  async member => {
    try {
      const data = load();

      const cfg =
        data.configs?.[
          member.guild.id
        ];

      if (
        !cfg?.leave?.enabled ||
        !cfg.leave.channelId
      ) {
        return;
      }

      const channel =
        member.guild.channels.cache.get(
          cfg.leave.channelId
        );

      if (
        !channel?.isTextBased()
      ) {
        return;
      }

      const text =
        cfg.leave.message
          .replaceAll(
            "<@user>",
            member.user.tag
          )
          .replaceAll(
            "{user}",
            member.user.tag
          )
          .replaceAll(
            "{username}",
            member.user.username
          )
          .replaceAll(
            "{server}",
            member.guild.name
          );

      await channel.send({
        content: text
      });
    } catch (error) {
      console.error(
        "❌ Erreur départ membre :",
        error
      );
    }
  }
);

client.on(
  "messageCreate",
  async message => {
    try {
      if (
        message.author.bot
      ) {
        return;
      }

      const data = load();

      // ===============================================
      // MESSAGES PRIVÉS
      // ===============================================

      if (
        !message.guild
      ) {
        const dmGuildIds =
          Object.keys(
            data.configs || {}
          );

        const dmAllowed =
          dmGuildIds.length === 0 ||
          dmGuildIds.some(
            guildId =>
              data.configs[guildId]
                ?.ai
                ?.dmEnabled !== false
          );

        if (!dmAllowed) {
          return;
        }

        const conversationId =
          `dm:${message.author.id}`;

        const conversation =
          getConversation(
            data,
            conversationId,
            "dm",
            message.author.id
          );

        const imageUrls =
          [
            ...message.attachments.values()
          ]
            .filter(
              attachment =>
                attachment.contentType
                  ?.startsWith("image/")
            )
            .map(
              attachment =>
                attachment.url
            );

        await queueAI(
          conversationId,
          async () => {
            addAIMessage(
              conversation,
              "user",
              message.author.id,
              message.author.username,
              message.content ||
              "[Image]"
            );

            await save(data);

            await message.channel
              .sendTyping()
              .catch(() => {});

            const response =
              await generateAIResponse(
                conversation,
                message.author,
                message.content,
                imageUrls,
                null
              );

            addAIMessage(
              conversation,
              "assistant",
              client.user.id,
              client.user.username,
              response
            );

            await save(data);

            await sendLong(
              message.channel,
              response,
              message
            );
          }
        );

        return;
      }

      // ===============================================
      // MESSAGES PERSONNALISÉS
      // ===============================================

      const cfg =
        data.configs?.[
          message.guild.id
        ];

      if (
        cfg?.customMessages?.length
      ) {
        const found =
          cfg.customMessages.find(
            item =>
              item.trigger
                ?.toLowerCase()
                .trim() ===
              message.content
                ?.toLowerCase()
                .trim()
          );

        if (found?.response) {
          await message.reply({
            content:
              found.response
          });

          return;
        }
      }

      // ===============================================
      // IA SERVEUR
      // ===============================================

      ensureAI(data);

      if (
        data.ai.channels[
          message.channel.id
        ] !== true
      ) {
        return;
      }

      const convId =
        activeConversation(
          data,
          message.guild.id,
          message.author.id
        );

      const conv =
        getConversation(
          data,
          convId,
          message.guild.id,
          message.author.id
        );

      // Si l'utilisateur n'est pas membre de la
      // conversation partagée, il reste en solo.
      if (
        conv.members.length > 1 &&
        !conv.members.includes(
          message.author.id
        )
      ) {
        return;
      }

      const imageUrls =
        [
          ...message.attachments.values()
        ]
          .filter(
            attachment =>
              attachment.contentType
                ?.startsWith("image/")
          )
          .map(
            attachment =>
              attachment.url
          );

      await queueAI(
        convId,
        async () => {
          addAIMessage(
            conv,
            "user",
            message.author.id,
            message.author.username,
            message.content ||
            "[Image]"
          );

          await save(data);

          await message.channel
            .sendTyping()
            .catch(() => {});

          const response =
            await generateAIResponse(
              conv,
              message.author,
              message.content,
              imageUrls,
              message.guild.name
            );

          addAIMessage(
            conv,
            "assistant",
            client.user.id,
            client.user.username,
            response
          );

          await save(data);

          await sendLong(
            message.channel,
            response,
            message
          );
        }
      );
    } catch (error) {
      console.error(
        "❌ Message IA :",
        error
      );

      await message.reply(
        "❌ Une erreur est survenue pendant que je réfléchissais."
      ).catch(() => {});
    }
  }
);
